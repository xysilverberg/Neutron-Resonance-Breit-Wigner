	\section{Pendahuluan}

	\begin{frame}{Tujuan Praktikum}
		\begin{exampleblock}{Tujuan}
			\begin{enumerate}
				\item Mengenal karakteristik dan sifat fisis penjalaran gelombang bunyi sebagai gelombang mekanik longitudinal.
				\item Memahami konsep taraf intensitas bunyi (dB) dan memformulasikannya sebagai fungsi kuadrat tegangan listrik \textit{input} transduser.
				\item Menggambar dan menganalisis kurva respon ambang dengar manusia pada kertas grafik semi-logaritmis untuk telinga kanan dan telinga kiri.
			\end{enumerate}
		\end{exampleblock}

		\begin{block}{Ringkasan Eksperimen}
			\footnotesize
			Pengukuran ambang dengar dilakukan pada 28 variasi frekuensi (20 Hz -- 20 kHz) menggunakan \textit{Digital Square Generator} (DSG) dan \textit{headphone} sebagai transduser. Nilai tegangan ambang ($V_{PP}$) diukur menggunakan osiloskop, sedangkan taraf intensitas direkam menggunakan \textit{Sound Level Meter} (SLM). Subjek uji adalah telinga kanan (Hendra Maulana) dan telinga kiri (Naufal Luthfian Hakim).
		\end{block}
	\end{frame}

	\begin{frame}{Latar Belakang}
		\begin{block}{Persepsi Bunyi dan Sistem Pendengaran Manusia}
			\footnotesize
			Sistem pendengaran manusia merupakan transduser biologis yang mengonversi gelombang tekanan mekanik menjadi sinyal saraf. Tanggapan fisiologis telinga bersifat non-linier dan sangat bergantung pada frekuensi, sehingga membutuhkan skala logaritmik (desibel) untuk merepresentasikannya secara kuantitatif \parencite{halliday2021}.
		\end{block}

		\begin{block}{Pentingnya Kajian Ambang Dengar}
			\footnotesize
			Kurva kontur kenyaringan setara ISO 226:2023 menunjukkan bahwa kepekaan telinga manusia tidak seragam di seluruh spektrum 20 Hz -- 20 kHz. Telinga paling sensitif pada rentang 1--4 kHz, di mana resonansi saluran telinga luar memaksimalkan transfer energi akustik ke membran timpani \parencite{moller2024}. Pemahaman ini penting dalam audiometri klinis, desain sistem audio, dan ergonomi akustik.
		\end{block}

		\begin{block}{Penemuan Kurva Ambang Dengar}
			\footnotesize
			Konsep intensitas dan taraf intensitas bunyi diformalkan oleh Harvey Fletcher dan Wilden A.\ Munson (1933) melalui eksperimen psikofisik yang menghasilkan kurva \textit{equal-loudness contour}, yang kemudian menjadi dasar standar ISO 226 \parencite{iso2023}.
		\end{block}
	\end{frame}

	\section{Dasar Teori}

	\begin{frame}{Dasar Teori: Persamaan Gelombang Mekanik (1)}
		\begin{block}{Persamaan Diferensial Gelombang Satu Dimensi}
			Gelombang mekanik longitudinal merambat melalui medium kontinu berdasarkan perpindahan energi tanpa perpindahan massa. Dinamika rambatan gelombang harmonik dinyatakan melalui persamaan diferensial satu dimensi \parencite{halliday2021}:
			\begin{equation}
				\frac{\partial^{2}Y}{\partial x^{2}} = \frac{1}{v^{2}}\,\frac{\partial^{2}Y}{\partial t^{2}}
			\end{equation}
			dengan $Y(x,t)$ simpangan partikel medium pada posisi $x$ dan waktu $t$, serta $v$ laju rambat gelombang.
		\end{block}

		\begin{block}{Persamaan Simpangan Partikel}
			Simpangan partikel medium akibat usikan gelombang sinusoidal:
			\begin{equation}
				Y = Y_{m}\sin(kx - \omega t)
			\end{equation}
			dengan $Y_{m}$ amplitudo simpangan maksimum, $k = 2\pi/\lambda$ bilangan gelombang, dan $\omega = 2\pi f$ frekuensi sudut.
		\end{block}
	\end{frame}

	\begin{frame}{Dasar Teori: Tekanan Akustik dan Intensitas Bunyi (2)}
		\begin{block}{Amplitudo Tekanan Puncak}
			Usikan partikel medium memicu fluktuasi tekanan akustik $p(x,t)$. Melalui modul elastisitas medium ($B=\rho_0 v^2$), amplitudo tekanan puncak ($p_m$) didefinisikan sebagai \parencite{halliday2021}:
			\begin{equation}
				p_{m} = Y_{m}\,k\,\rho_{0}\,v^{2}
			\end{equation}
			Fluktuasi tekanan berbanding lurus dengan amplitudo simpangan, frekuensi, massa jenis, dan kuadrat laju rambat bunyi.
		\end{block}

		\begin{block}{Intensitas Gelombang Bunyi}
			Laju fluks energi akustik yang menembus satuan luas per satuan waktu mendefinisikan intensitas bunyi ($I$):
			\begin{equation}
				I = 2\pi^{2}\rho\,v\,Y_{m}^{2}\,f^{2}
			\end{equation}
			di mana $\rho$ kerapatan medium, $v$ kecepatan bunyi, $Y_m$ amplitudo simpangan, dan $f$ frekuensi gelombang.
		\end{block}
	\end{frame}

	\begin{frame}{Dasar Teori: Taraf Intensitas Bunyi (3)}
		\begin{block}{Taraf Intensitas dalam Skala Desibel}
			Tanggapan fisiologis telinga yang non-linier direpresentasikan dalam skala logaritmik. Taraf intensitas bunyi ($\beta$) dalam desibel (dB) dinyatakan dengan membandingkan intensitas terukur ($I$) terhadap $I_0 = 10^{-12}\ \text{W/m}^2$ \parencite{halliday2021}:
			\begin{equation}
				\beta = 10\,\log_{10}\!\left(\frac{I}{I_0}\right)
			\end{equation}
		\end{block}

		\begin{block}{Taraf Intensitas Berbasis Tegangan Listrik}
			Karena daya mekanik akustik transduser berbanding lurus dengan kuadrat tegangan masukan ($P \propto V^2$), taraf intensitas bunyi relatif ($\beta$) dapat dihitung melalui rasio tegangan puncak-ke-puncak \parencite{ward2022}:
			\begin{equation}
				\beta = 20\,\log_{10}\!\left(\frac{V}{V_0}\right)
			\end{equation}
			dengan $V$ tegangan masukan pada frekuensi uji dan $V_0$ tegangan referensi pada ambang dengar standar.
		\end{block}
	\end{frame}

	\begin{frame}{Dasar Teori: Model Taraf Intensitas Teoritis (4)}
		\begin{block}{Taraf Intensitas Teoritis dari Parameter Kalibrasi}
			Dengan mengintegrasikan hasil kalibrasi titik referensi (1 kHz) ke dalam Persamaan (6), taraf intensitas teoritis dinyatakan sebagai \parencite{zhao2023}:
			\begin{equation}
				TI_{\text{teori}} = TI_0 + 20\,\log_{10}\!\left(\frac{V_{PP}}{V_0}\right)
			\end{equation}
			dengan $TI_0$ taraf intensitas pada 1 kHz, $V_{PP}$ amplitudo tegangan puncak-ke-puncak, dan $V_0$ tegangan ambang referensi.
		\end{block}

		\begin{block}{Selisih Terukur dan Teoritis}
			\begin{equation}
				\Delta TI = TI_{\text{rata}^2} - TI_{\text{teori}}
			\end{equation}
		\end{block}

		\begin{block}{Root Mean Square Error (RMSE)}
			\begin{equation}
				\text{RMSE} = \sqrt{\frac{1}{N}\sum_{i=1}^{N}(\Delta TI_i)^{2}}
			\end{equation}
		\end{block}
	\end{frame}

	\begin{frame}{Alat dan Bahan}
		\begin{columns}[T]
			\column{0.55\textwidth}
			\begin{block}{Alat}
				\footnotesize
				\begin{itemize}
					\item \textbf{Digital Square Generator (DSG):} pembangkit sinyal persegi digital dan pengatur amplitudo.
					\item \textbf{Headphone Audio:} transduser elektroakustik pengubah daya listrik menjadi energi mekanik.
					\item \textbf{Osiloskop Digital:} pengukur nilai tegangan puncak-ke-puncak ($V_{PP}$).
					\item \textbf{Sound Level Meter (SLM):} alat ukur taraf intensitas bunyi dalam satuan dB.
					\item \textbf{Kabel Penghubung BNC/Audio:} penghubung interkoneksi jalur sinyal.
				\end{itemize}

			\begin{block}{Subjek Uji (Testi)}

				\end{itemize}
			\end{block}

			\begin{exampleblock}{Variabel Eksperimen}
				\footnotesize
				\begin{itemize}
					\item 28 titik frekuensi: $20\ \text{Hz}$ -- $20\ \text{kHz}$, spasi logaritmik.
					\item Titik referensi kalibrasi: $f_0 = 1\ \text{kHz}$.
					\item Pengulangan: 3 kali pembacaan $TI$ per frekuensi.
				\end{itemize}

	\section{Pembahasan}

	\begin{frame}{Pembahasan (1): Karakteristik Ambang Dengar}
		\begin{itemize}
			\item \textbf{Kepekaan Tidak Seragam terhadap Frekuensi:}
			Ambang dengar manusia terbukti tidak seragam di seluruh spektrum 20 Hz--20 kHz. $TI_{\min}$ telinga kanan ($51{,}53$ dB pada 2 kHz) bersesuaian dengan kurva ISO 226:2023 dan kurva Fletcher--Munson \parencite{iso2023}.

			\vspace{0.15cm}
			\item \textbf{Resonansi Saluran Telinga Luar:}
			Sensitivitas maksimum pada 2--4 kHz disebabkan oleh efek resonansi saluran telinga luar (\textit{ear canal resonance}). Saluran telinga berperan sebagai pipa organa tertutup satu ujung (panjang $\sim\!2{,}5$ cm) dengan frekuensi resonansi alami sekitar 2{,}5--4 kHz, memaksimalkan transfer daya akustik ke membran timpani \parencite{moller2024}.

			\vspace{0.15cm}
			\item \textbf{Hubungan dengan Hukum Taraf Intensitas (Pers.\ 5 dan 6):}
			Skala desibel digunakan karena telinga manusia merespons intensitas secara logaritmik ($\beta \propto \log I$), sehingga rentang dinamisnya yang sangat lebar ($10^{-12}$ hingga $10^{2}\ \text{W/m}^2$) dapat direpresentasikan dalam rentang 0--130 dB.
		\end{itemize}
	\end{frame}

	\begin{frame}{Pembahasan (2): Asimetri Interaural dan Evaluasi Model}
		\begin{itemize}
			\item \textbf{Asimetri Interaural:}
			Meskipun rata-rata $TI$ kedua telinga berdekatan (59{,}49 dB vs.\ 60{,}10 dB), terdapat perbedaan signifikan pada tegangan referensi ($V_0 = 2$ mV vs.\ $7$ mV). Asimetri ini umum dalam audiometri subjektif, dipengaruhi oleh variasi kelelahan pendengaran sesaat, perbedaan geometris liang telinga, tingkat atensi psikofisik, dan kekedapan bantalan \textit{headphone} \parencite{ward2022}.

			\vspace{0.15cm}
			\item \textbf{Deviasi Model Teoritis --- Penyebab Selisih Negatif:}
			RMSE $= 13{,}902$ dB (kanan) dan $18{,}890$ dB (kiri) dengan $\overline{\Delta TI}$ negatif konsisten disebabkan oleh:
			\begin{enumerate}
				\footnotesize
				\item \textbf{Respon frekuensi headphone tidak linier:} efisiensi konversi daya listrik ke tekanan akustik bervariasi di seluruh 20 Hz--20 kHz.
				\item \textbf{Kebocoran akustik SLM:} sensor di luar liang telinga mengalami atenuasi berbeda dari tekanan nyata di membran timpani \parencite{ward2022}.
				\item \textbf{Batas derau ruangan:} SLM memiliki batas sekitar 45--55 dB \parencite{timdosen2026}.
			\end{enumerate}

			\vspace{0.1cm}
			\item \textbf{Validitas Data:}
			Meskipun terdapat deviasi, model teoritis mampu menangkap tren perubahan $TI$ terhadap variasi amplitudo input. Hasil pengukuran tetap valid untuk analisis karakteristik ambang dengar.
		\end{itemize}
	\end{frame}

				\item Gelombang bunyi merambat sebagai gelombang mekanik longitudinal; intensitas akustik sebanding dengan kuadrat amplitudo tegangan listrik masukan transduser.
				\item Kepekaan ambang dengar manusia tidak linier terhadap frekuensi, dengan sensitivitas maksimum pada 1--2 kHz selaras dengan kurva ISO 226.
				\item Telinga kanan: $TI_{\min} = 51{,}53$ dB pada 2 kHz ($V_0=2$ mV); telinga kiri: $TI_{\min} = 51{,}87$ dB pada 20 kHz serta $52{,}50$ dB pada 1 kHz ($V_0=7$ mV).

			\end{enumerate}
		\end{block}

		\begin{block}{Saran untuk Praktikum Selanjutnya}

		\tiny
		\begin{columns}[T]
			\column{0.48\textwidth}
			\textbf{Persamaan Diferensial Gelombang}
			\begin{equation*}
				\frac{\partial^{2}Y}{\partial x^{2}} = \frac{1}{v^{2}}\,\frac{\partial^{2}Y}{\partial t^{2}}
			\end{equation*}

			\textbf{Simpangan Partikel}
			\begin{equation*}
				Y = Y_{m}\sin(kx - \omega t)
			\end{equation*}

			\textbf{Amplitudo Tekanan Puncak}
			\begin{equation*}
				p_{m} = Y_{m}\,k\,\rho_{0}\,v^{2}
			\end{equation*}

			\textbf{Intensitas Gelombang Bunyi}
			\begin{equation*}
				I = 2\pi^{2}\rho\,v\,Y_{m}^{2}\,f^{2}
			\end{equation*}

			\vspace{0.15cm}
			\textbf{Taraf Intensitas Bunyi (dB)}
			\begin{equation*}
				\beta = 10\,\log_{10}\!\left(\frac{I}{I_0}\right), \quad I_0 = 10^{-12}\ \text{W/m}^2
			\end{equation*}

			\column{0.48\textwidth}
			\textbf{Taraf Intensitas Berbasis Tegangan}
			\begin{equation*}
				\beta = 20\,\log_{10}\!\left(\frac{V}{V_0}\right)
			\end{equation*}

			\textbf{Model Taraf Intensitas Teoritis}
			\begin{equation*}
				TI_{\text{teori}} = TI_0 + 20\,\log_{10}\!\left(\frac{V_{PP}}{V_0}\right)
			\end{equation*}

			\textbf{Selisih $TI$ Terukur dan Teoritis}
			\begin{equation*}
				\Delta TI = TI_{\text{rata}^2} - TI_{\text{teori}}
			\end{equation*}

			\textbf{Root Mean Square Error (RMSE)}
			\begin{equation*}
				\text{RMSE} = \sqrt{\frac{1}{N}\sum_{i=1}^{N}(\Delta TI_i)^{2}}
			\end{equation*}

			\vspace{0.15cm}
			\textbf{Hasil:}\\
			Telinga Kanan: $\text{RMSE} = 13{,}902\ \text{dB}$\\
